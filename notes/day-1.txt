binding:
	data: passing data from code of the component to the template of the same component
		syntax: 
		a. interpolation: {{ prop-name}} 
		note: value is stringified before being rendered

		b. property binding:
		<element [attr]="prop-name">
		note: value is NOT stringified before being rendered

	event: passing data from the template of the component to the code of the same component
		syntac: <elemnt (event-name)="method-in-code(args...)">

change detection:
	angular detects the change in the properties of the component and accordingly updates every element in the template where the same is bound

	exports: [NgModelDirective]

export class FormsModule{

}

@Directive({
	selector:'[ngModel]',
	standalone:false
})
export class NgModelDirective{
	function(first){
		const ctrl = new FormControl()
		ctrl.value = first
		ctrl.addEventHandler('valueChanges', function(e){
			ctrl.value = e.target.value
		})
	}
}
